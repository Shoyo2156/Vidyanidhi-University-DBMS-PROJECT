CREATE DATABASE vidyanidhi_university;
USE vidyanidhi_university;

CREATE TABLE college (
    college_id INT PRIMARY KEY AUTO_INCREMENT,
    college_name VARCHAR(100) NOT NULL,
    address VARCHAR(200)
);

CREATE TABLE degree (
    degree_id INT PRIMARY KEY AUTO_INCREMENT,
    degree_name VARCHAR(100) NOT NULL,
    duration INT
);

CREATE TABLE student (
    exam_seat_no INT PRIMARY KEY,
    student_name VARCHAR(100) NOT NULL,
    college_id INT,
    degree_id INT,
    gender VARCHAR(10),
    mobile_no VARCHAR(15),
    email VARCHAR(100),
    FOREIGN KEY (college_id) REFERENCES college(college_id),
    FOREIGN KEY (degree_id) REFERENCES degree(degree_id)
);

CREATE TABLE subject (
    subject_id INT PRIMARY KEY AUTO_INCREMENT,
    subject_name VARCHAR(100) NOT NULL,
    degree_id INT,
    credits INT,
    FOREIGN KEY (degree_id) REFERENCES degree(degree_id)
);

CREATE TABLE subject_section (
    section_id INT PRIMARY KEY AUTO_INCREMENT,
    subject_id INT,
    section_name VARCHAR(50),
    FOREIGN KEY (subject_id) REFERENCES subject(subject_id)
);

CREATE TABLE exam_center (
    center_id INT PRIMARY KEY AUTO_INCREMENT,
    center_name VARCHAR(100),
    address VARCHAR(200),
    capacity INT
);

CREATE TABLE exam_schedule (
    schedule_id INT PRIMARY KEY AUTO_INCREMENT,
    subject_id INT,
    center_id INT,
    exam_date DATE,
    exam_time TIME,
    total_students INT,
    FOREIGN KEY (subject_id) REFERENCES subject(subject_id),
    FOREIGN KEY (center_id) REFERENCES exam_center(center_id)
);

CREATE TABLE student_subject (
    student_subject_id INT PRIMARY KEY AUTO_INCREMENT,
    exam_seat_no INT,
    subject_id INT,
    FOREIGN KEY (exam_seat_no) REFERENCES student(exam_seat_no),
    FOREIGN KEY (subject_id) REFERENCES subject(subject_id)
);

CREATE TABLE result (
    result_id INT PRIMARY KEY AUTO_INCREMENT,
    exam_seat_no INT,
    subject_id INT,
    marks_obtained DECIMAL(5,2),
    grade CHAR(2),
    status VARCHAR(20),
    FOREIGN KEY (exam_seat_no) REFERENCES student(exam_seat_no),
    FOREIGN KEY (subject_id) REFERENCES subject(subject_id)
);

CREATE TABLE exam_fee (
    fee_id INT PRIMARY KEY AUTO_INCREMENT,
    exam_seat_no INT,
    amount DECIMAL(10,2),
    payment_date DATE,
    payment_status VARCHAR(20),
    FOREIGN KEY (exam_seat_no) REFERENCES student(exam_seat_no)
);

CREATE TABLE expenditure (
    expenditure_id INT PRIMARY KEY AUTO_INCREMENT,
    center_id INT,
    expense_type VARCHAR(100),
    amount DECIMAL(10,2),
    expense_date DATE,
    FOREIGN KEY (center_id) REFERENCES exam_center(center_id)
);

SELECT s.student_name,
       sub.subject_name,
       r.marks_obtained,
       r.grade
FROM student s
JOIN result r
ON s.exam_seat_no = r.exam_seat_no
JOIN subject sub
ON r.subject_id = sub.subject_id;

SELECT sub.subject_name,
       COUNT(ss.exam_seat_no) AS total_students
FROM subject sub
JOIN student_subject ss
ON sub.subject_id = ss.subject_id
GROUP BY sub.subject_name;

SELECT center_name,total_students
FROM exam_schedule es
JOIN exam_center ec
ON es.center_id = ec.center_id;

SELECT s.student_name,
       r.marks_obtained
FROM student s
JOIN result r
ON s.exam_seat_no = r.exam_seat_no
ORDER BY r.marks_obtained DESC;

SELECT SUM(amount) AS total_collection
FROM exam_fee
WHERE payment_status='Paid';

SELECT s.student_name,
       ef.amount
FROM student s
JOIN exam_fee ef
ON s.exam_seat_no=ef.exam_seat_no
WHERE ef.payment_status='Pending';

SELECT SUM(amount) AS total_expenditure
FROM expenditure;

SELECT
(
    SELECT SUM(amount)
    FROM exam_fee
    WHERE payment_status='Paid'
)
-
(
    SELECT SUM(amount)
    FROM expenditure
) AS profit_or_loss;
